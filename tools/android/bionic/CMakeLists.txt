cmake_minimum_required(VERSION 3.24)
project(gow3_bionic_driver_check LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Use the Android NDK toolchain with ANDROID_ABI=arm64-v8a")
endif()
# The driver loader is independent of Vulkan headers and never links libvulkan.
add_executable(gow3-vulkan-check driver_probe.cpp ../../../gpu/shim/vk_driver_loader.cpp)
target_include_directories(gow3-vulkan-check PRIVATE ../../../gpu/shim)
target_link_libraries(gow3-vulkan-check PRIVATE dl)
target_compile_options(gow3-vulkan-check PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
add_executable(kgsl-chip-id ../build/kgsl_chip_id.c)
target_compile_options(kgsl-chip-id PRIVATE -Wall -Wextra -Werror)
install(TARGETS gow3-vulkan-check kgsl-chip-id RUNTIME DESTINATION bin)
